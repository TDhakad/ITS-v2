CREATE TABLE projects (
	id INTEGER NOT NULL, 
	name VARCHAR(120) NOT NULL, 
	slug VARCHAR(80) NOT NULL, 
	description TEXT NOT NULL, 
	owner_id INTEGER, 
	is_active BOOLEAN NOT NULL, 
	created_at DATETIME NOT NULL, 
	PRIMARY KEY (id), 
	UNIQUE (name), 
	FOREIGN KEY(owner_id) REFERENCES users (id) ON DELETE SET NULL
)

Column Details:
-- id: Primary key for the project record.
-- name: Unique human-friendly project name.
-- slug: Unique URL-safe project identifier.
-- description: Project description shown in management views.
-- owner_id: Owning user id; nullable when owner is removed.
-- is_active: Soft-enable flag for project availability.
-- created_at: UTC timestamp when the project was created.
